The Space orange with white is 3.33:1: fine for the 24dp badge and bold 14sp+ chips, below AA for small regular text, so the chip label is always 12sp/700 or larger and never used for body copy. @Immutable
data class CinemerickColors(
    val background: Color, val surface: Color,
    val surfaceContainer: Color, val surfaceContainerHigh: Color,
    val outlineVariant: Color, val onSurface: Color,
    val onSurfaceVariant: Color, val primary: Color,
    val onPrimary: Color, val primaryContainer: Color,
    val onPrimaryContainer: Color, val scrim: Color,
    val chainEdge: Color,
)
val DarkColors = CinemerickColors(
    background = Color(0xFF0D0B14), surface = Color(0xFF15121F),
    surfaceContainer = Color(0xFF1D1930),
    surfaceContainerHigh = Color(0xFF262140),
    primary = Color(0xFFA78BFA), onPrimary = Color(0xFF1A0F3D),
    primaryContainer = Color(0xFF3A2C7A),
    onPrimaryContainer = Color(0xFFE7DEFF), /* … */
)
val LocalCinemerickColors =
    staticCompositionLocalOf<CinemerickColors> { DarkColors }

object CinemerickTheme {
    val colors @Composable get() = LocalCinemerickColors.current
    val spacing = Spacing()   // space1..space12
    val chain = CinemaBrandColors   // existing object, untouched
}
